+    return node.value

+def _keywords(call):

+def _update_dict(tree, receiver):

+            continue
+        if node.func.attr == "update" and ast.unparse(node.func.value) == receiver:
+            assert len(node.args) == 1 and isinstance(node.args[0], ast.Dict)
